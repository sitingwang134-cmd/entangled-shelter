        var archiveRecords = [];
        var archiveNotes = {};
        var archiveFavourite = "";
        var archiveBank = 0;
        var archiveKey = 'entangled-shelter:decisions:v1';
        var labels = {food:'TASTE',style:'STYLE',travel:'WANDER'};
        var modeLabels = {poker:'FORTUNE DECK',rps:'A FAIR FIGHT',guess:'TRUST YOUR HUNCH'};
        function readArchive() {
          try {
            var data = JSON.parse(localStorage.getItem(archiveKey) || '[]');
            archiveRecords = Array.isArray(data) ? data.filter(function(r) {
              return r && typeof r.id === 'string' && Object.hasOwn(labels,r.categoryId) &&
                Object.hasOwn(modeLabels,r.mode) && r.option && typeof r.option.name === 'string' &&
                typeof r.option.detail === 'string' && typeof r.decidedAt === 'string' && Number.isFinite(Date.parse(r.decidedAt));
            }).slice(0,500) : [];
          } catch (_) { archiveRecords = []; }
          try { archiveNotes=JSON.parse(localStorage.getItem('entangled-shelter:notes:v1')||'{}'); if(!archiveNotes || typeof archiveNotes!=='object') archiveNotes={}; } catch (_) {archiveNotes={};}
          var counts={}; archiveRecords.forEach(function(r){var key=r.categoryId+':'+r.option.id; if(!counts[key]) counts[key]={name:r.option.name,count:0};counts[key].count++;});
          var ranked=Object.values(counts).sort(function(a,b){return b.count-a.count;});
          archiveFavourite=ranked.length ? ranked[0].name+' / '+ranked[0].count+' TIMES' : 'YOUR STORY IS STILL TO COME';
          archiveBank = Math.min(archiveBank, Math.max(0, Math.ceil(archiveRecords.length / 8) - 1));
        }
        function fullTime(value) {
          return new Date(value).toLocaleString('en-GB', {year:'numeric',month:'short',day:'2-digit',hour:'2-digit',minute:'2-digit',hour12:false});
        }
        var photoCache = {};
        var refreshPhotoAtlas = null;
        var photoRefreshPending = false;
        var folioPhotos = [
          ['breakfast','TASTE / SLOW MORNINGS'],
          ['coast','WANDER / COLOUR BY THE SEA'],
          ['boutique','STYLE / THE DAILY EDIT'],
          ['pasta','TASTE / BEFORE THE SUPPER'],
          ['alps','WANDER / STILL WATERS'],
          ['shoes','STYLE / STEP OUTSIDE'],
          ['coffee','TASTE / A MOMENT TO PAUSE'],
          ['station','WANDER / THE NEXT DEPARTURE']
        ];
        function photoFor(record, index) {
          if (record) {
            var choices = {food:['breakfast','pasta','coffee'],style:['boutique','shoes'],travel:['coast','alps','station']}[record.categoryId];
            return choices[index % choices.length];
          }
          return folioPhotos[index % folioPhotos.length][0];
        }
        function photograph(id) {
          if (!photoCache[id]) {
            var img = new Image(); photoCache[id] = img;
            img.onload = function() {
              if (photoRefreshPending) return;
              photoRefreshPending = true;
              requestAnimationFrame(function() {
                photoRefreshPending = false;
                if (refreshPhotoAtlas) refreshPhotoAtlas();
              });
            };
            img.src = '/entangled/roll-photos/' + id + '.webp';
          }
          return photoCache[id];
        }
        function coverPhoto(g, img, x, y, w, h) {
          if (!img.complete || !img.naturalWidth) return;
          var scale = Math.max(w/img.naturalWidth,h/img.naturalHeight);
          var sw=w/scale, sh=h/scale;
          g.drawImage(img,(img.naturalWidth-sw)/2,(img.naturalHeight-sh)/2,sw,sh,x,y,w,h);
        }
        readArchive();
        function buildAtlas() {
          var CELL = 512;
          var cv = document.createElement('canvas');
          cv.width = CELL * ATLAS_N;
          cv.height = CELL;
          var g = cv.getContext('2d');
          g.fillStyle = '#f6f3e9';
          g.fillRect(0,0,cv.width,cv.height);
          for(var grain=0;grain<2600;grain++) {
            g.fillStyle='rgba(120,116,105,'+(0.015+rand()*0.03)+')';
            g.fillRect(rand()*cv.width,rand()*cv.height,1+rand()*2,1);
          }
          var bank = archiveRecords.slice(archiveBank * 8, archiveBank * 8 + 8);
          function line(text,x,y,width,font) {
            g.font=font; var size=parseInt(font,10);
            while(g.measureText(text).width > width && size > 13) { size--; g.font=size+'px Georgia, serif'; }
            g.fillText(text,x,y,width);
          }
          for(var c=0;c<ATLAS_N;c++) {
            var x=c*CELL; var r=bank.length ? bank[c % bank.length] : null;
            var id=photoFor(r,c);
            // Full-width photographic panel makes the imagery readable on the curved paper.
            g.fillStyle='#d8ded3'; g.fillRect(x+18,18,476,330);
            coverPhoto(g,photograph(id),x+18,18,476,330);
            var shade=g.createLinearGradient(0,220,0,348);
            shade.addColorStop(0,'rgba(15,28,22,0)'); shade.addColorStop(1,'rgba(15,28,22,.78)');
            g.fillStyle=shade;g.fillRect(x+18,220,476,128);
            g.fillStyle='#fff9e9';
            line('ES / '+(r ? labels[r.categoryId] : folioPhotos[c%folioPhotos.length][1].split(' / ')[0]),x+36,53,430,'19px Georgia, serif');
            line(r ? r.option.name : folioPhotos[c%folioPhotos.length][1].split(' / ')[1],x+36,324,430,'32px Georgia, serif');
            g.fillStyle='#243029';
            line(r ? modeLabels[r.mode] : 'THE POSSIBILITY COLLECTION',x+30,382,452,'17px Georgia, serif');
            line(r && typeof archiveNotes[r.id]==='string' && archiveNotes[r.id] ? archiveNotes[r.id] : r ? r.option.detail : typeof archiveNotes.reflection==='string' && archiveNotes.reflection ? archiveNotes.reflection : 'A little chance. A little courage.',x+30,415,452,'19px Georgia, serif');
            g.fillRect(x+30,438,452,1);
            line(r ? 'MOST RETURNED TO: '+archiveFavourite : archiveFavourite,x+30,464,452,'14px Georgia, serif');
            line(r ? 'DECISION '+String(archiveBank*8+c%bank.length+1).padStart(4,'0') : 'FOLIO '+String(c+1).padStart(2,'0')+' / ENTANGLED SHELTER',x+30,490,452,'13px Georgia, serif');
          }
          var tex=new THREE.CanvasTexture(cv);
          tex.wrapS=THREE.RepeatWrapping;
          tex.wrapT=THREE.ClampToEdgeWrapping;
          tex.encoding=THREE.sRGBEncoding;
          tex.anisotropy=renderer.capabilities.getMaxAnisotropy();
          return tex;
        }
        function installArchiveControls() {
          var list=document.getElementById('recordList');
          var panel=document.getElementById('records');
          var toggle=document.getElementById('listToggle');
          var next=document.getElementById('bankButton');
          function refreshTexture() {
            var nextTexture=buildAtlas();
            atlasTex.image=nextTexture.image; atlasTex.needsUpdate=true;
            barrelTex.image=nextTexture.image; barrelTex.needsUpdate=true;
            nextTexture.dispose();
            next.hidden=archiveRecords.length<=8;
            next.textContent='FOLIO '+(archiveBank+1)+' / '+Math.max(1,Math.ceil(archiveRecords.length/8))+' ↻';
          }
          function renderList() {
            list.replaceChildren();
            document.getElementById('recordCount').textContent=archiveRecords.length ? archiveRecords.length+' decisions preserved · newest first' : 'No decisions yet. Return to the shelter and accept a result to leave your first trace.';
            archiveRecords.forEach(function(r,index) {
              var li=document.createElement('li');
              var category=document.createElement('p'); category.textContent=labels[r.categoryId]+' · '+modeLabels[r.mode];
              var name=document.createElement('strong'); name.textContent=r.option.name;
              var detail=document.createElement('p'); detail.textContent=r.option.detail;
              var time=document.createElement('time'); time.dateTime=r.decidedAt; time.textContent=fullTime(r.decidedAt);
              var view=document.createElement('button'); view.textContent='Print on the roll ↗';
              view.addEventListener('click',function(){archiveBank=Math.floor(index/8);refreshTexture();panel.hidden=true;toggle.setAttribute('aria-expanded','false');setPaused(false);toggle.focus();});
              var note=document.createElement('p'); note.textContent=typeof archiveNotes[r.id]==='string' ? archiveNotes[r.id] : '';
              li.append(category,name,detail,note,time,view);list.append(li);
            });
          }
          toggle.addEventListener('click',function(){panel.hidden=!panel.hidden;toggle.setAttribute('aria-expanded',String(!panel.hidden));setPaused(!panel.hidden);});
          next.addEventListener('click',function(){archiveBank=(archiveBank+1)%Math.max(1,Math.ceil(archiveRecords.length/8));refreshTexture();});
          window.addEventListener('keydown',function(e){if(e.key==='Escape'&&!panel.hidden){panel.hidden=true;toggle.setAttribute('aria-expanded','false');setPaused(false);toggle.focus();}else if(e.code==='Space' && e.target===document.body){e.preventDefault();setPaused(!paused);}});
          window.addEventListener('storage',function(e){if(e.key===archiveKey||e.key==='entangled-shelter:notes:v1'||e.key===null){readArchive();refreshTexture();renderList();}});
          document.getElementById('stage').addEventListener('webglcontextlost',function(e){e.preventDefault();document.getElementById('hintText').textContent='The scene paused unexpectedly. Refresh to restore it; your decisions remain available.';});
          refreshPhotoAtlas=refreshTexture;
          refreshTexture();renderList();
        }
